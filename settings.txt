# CellAppCP3 参数配置
# 修改后保存即可，下次运行自动读取

# ═══════════════════════════════════════
#  Cellpose 模型参数
# ═══════════════════════════════════════

# 细胞直径（像素）
# 作用：告诉模型细胞大概多大，影响分割精度
# 范围：50-300，太小会把一个细胞切成多个，太大会把多个细胞合成一个
# 建议：先量一下图里典型细胞的像素直径
diameter = 120

# 流场一致性阈值
# 作用：控制分割边界的严格程度，值越低接受越不规则的形状
# 范围：0.1-1.0，默认 0.4（v4）
# 调整：漏检多 → 调低；误检多 → 调高
flow_threshold = 0.95

# 细胞概率阈值
# 作用：判定像素属于细胞的门槛，值越低检测到的细胞越多
# 范围：-6.0 到 6.0，默认 0.0（v4）
# 调整：暗细胞漏检 → 调低（如 -2.0）；噪点太多 → 调高
cellprob_threshold = 1.0

# 最小细胞面积（像素）
# 作用：小于此面积的分割结果直接丢弃
# 范围：50-1000
# 调整：碎片噪点多 → 调高
min_size = 200

# 流场迭代次数
# 作用：影响分割精度和速度，次数越多越精确但越慢
# 范围：50-400，默认 200
# 调整：想加速 → 降到 100，精度损失很小
niter = 200

# Cellpose 模型名称
# 可选：cyto3（快，精度一般）/ cpsam（慢，精度高，需 1.2GB 模型文件）
# 本地有同名文件时优先加载本地，否则使用内置模型
model_name = cpsam


# ═══════════════════════════════════════
#  过滤参数（筛掉不合格的分割结果）
# ═══════════════════════════════════════

# 凸包完整度阈值
# 作用：mask面积 / 凸包面积，衡量细胞形状有无凹陷
# 范围：0.0-1.0，完整凸形接近 1.0
# 调整：正常细胞被过滤 → 调低；碎片通过太多 → 调高
hull_comp = 0.85

# 圆形完整度阈值
# 作用：mask面积 / 最小外接圆面积，衡量细胞有多圆
# 范围：0.0-1.0，完美圆形接近 1.0
# 调整：六边形/椭圆细胞被过滤 → 调低（如 0.4）
circle_comp = 0.65

# 暗区过滤阈值
# 作用：细胞平均亮度低于此值直接丢弃（滤掉黑边误识别）
# 范围：0-100
# 调整：暗细胞被误删 → 调低
dark_threshold = 15

# 面积过滤比例
# 作用：面积小于「所有细胞面积中位数 × 此比例」的丢弃
# 范围：0.05-0.5
# 调整：小细胞被误删 → 调低
area_ratio = 0.15

# 圆形度阈值
# 作用：4π×面积/周长²，衡量轮廓规则程度
# 范围：0.0-1.0，完美圆形 = 1.0
# 调整：不规则细胞被过滤 → 调低
circularity = 0.5

# 亮度计算最少像素数
# 作用：细胞内可用像素少于此数则跳过亮度计算
# 范围：20-200
min_pixels = 50

# 亮度取样百分比
# 作用：取细胞内最亮的前 N% 像素求平均作为亮度值
# 范围：0.01-0.2
# 调整：想反映整体亮度 → 调高（如 0.2）；只看峰值 → 调低
brightness_top_pct = 0.05


# ═══════════════════════════════════════
#  排行功能开关（可独立开启/关闭/同时开启）
# ═══════════════════════════════════════

# 最亮细胞排行数量
# 作用：仅在 enable_top_ranking = 1 时生效，绿色高亮前 N 个最亮细胞
# 注意：不影响破损排行（破损细胞会全部列出）
# 范围：1-50
top_n = 3

# 最亮细胞排行
# 1 = 开启：绿色高亮前 top_n 个最亮细胞 + 输出 crop/top/ 裁剪图
# 0 = 关闭：不做亮度排行（正常细胞仍用黄色标注）
enable_top_ranking = 1

# 破损细胞排行
# 1 = 开启：按暗块面积降序列出所有破损细胞 + 输出 crop/bad/ 全部裁剪图 + {图片名}_bad.csv
# 0 = 关闭：不做破损排行（破损细胞仍会红色标注，取决于 enable_bad_detection）
enable_bad_ranking = 1

# 排序方向（仅影响最亮排行）
# 1 = 从亮到暗（荧光图，最亮的排第一）
# 0 = 从暗到亮（亮场图，最暗的排第一，通常是有菌细胞）
sort_descending = 1


# ═══════════════════════════════════════
#  图片缩放
# ═══════════════════════════════════════

# 缩放倍率
# 作用：所有图片按此倍率等比例缩小后再识别
# 范围：0.1-1.0，1.0 = 不缩放
# 参考：20664x8232 的大图用 0.15 缩到约 3100x1235
# 调整：速度太慢 → 调低；精度不够 → 调高
resize_scale = 0.15


# ═══════════════════════════════════════
#  坏细胞检测（虫蛀 / 破损 / 空洞）
# ═══════════════════════════════════════

# 坏细胞检测总开关
# 1 = 开启，0 = 关闭
enable_bad_detection = 1

# 暗块检测阈值模式
# 0 = 固定阈值（用 bad_dark_threshold）
# 1 = 自适应阈值（按细胞自身内圈亮度计算）
bad_adaptive_threshold = 0

# 自适应阈值基准分位数
# 作用：用细胞内圈亮度的第 N 百分位作为基准，避免暗斑拉低平均值
# 50 = 中位数，75 = 上四分位（代表细胞正常区域亮度）
# 范围：50-90
# 调整：漏检 → 调高；误判 → 调低
bad_adaptive_percentile = 75

# 自适应阈值系数
# 作用：阈值 = 细胞内圈第 N 分位亮度 × 此系数
# 范围：0.3-0.7
# 调整：斑点漏检 → 调高；暗环误判 → 调低
bad_adaptive_ratio = 0.5

# 自适应模式基准亮度内圈比例
# 作用：只用细胞内圈像素算平均亮度，排除边缘暗环干扰
# 范围：0.5-0.8
bad_adaptive_inner = 0.6

# 固定阈值（bad_adaptive_threshold=0 时生效，或像素数不足时兜底）
# 作用：亮度低于此值的像素视为暗像素
# 范围：0-255
bad_dark_threshold = 75

# 边缘细胞排除距离（像素）
# 作用：距图像边界此范围内的细胞跳过检测
# 原因：贴边细胞轮廓被截断，会产生假缺口
# 范围：0-100
bad_edge_margin = 20

# 暗块最小面积（占细胞面积比例）
# 作用：小于此比例的暗块视为噪点忽略
# 范围：0.005-0.05
bad_blob_min_area = 0.01

# 紧凑度阈值（暗块面积 / 外接矩形面积）
# 作用：区分「环状暗环」和「团块状虫蛀」
#      环状暗环紧凑度低（细长包围），团块紧凑度高（接近圆形）
# 范围：0.3-0.7
# 调整：暗环被误判 → 调高；虫蛀漏检 → 调低
bad_blob_compactness = 0.35

# 暗块位置分界（暗块中心到细胞中心距离 / 细胞半径）
# 作用：区分「中心虫蛀」和「边缘暗环加厚」
# 小于此值视为中心区，用较低面积门槛；大于此值视为边缘区，用较高门槛
# 范围：0.3-0.7
bad_blob_center_radius = 0.5

# 中心区暗块面积门槛（占细胞面积比例）
# 作用：位于中心区的暗块，面积超过此值判为 HOLE
# 范围：0.003-0.02
# 调整：小虫蛀漏检 → 调低
bad_hole_area_center = 0.008

# 边缘区暗块面积门槛（占细胞面积比例）
# 作用：位于边缘区的暗块，面积超过此值才判为 HOLE
# 原因：细胞边缘暗环局部加厚会形成假暗块，需要更高门槛
# 范围：0.015-0.05
# 调整：暗环误判 → 调高；边缘真实虫蛀漏检 → 调低
bad_hole_area_edge = 0.035

# BROKEN 判定阈值（最大暗块面积占细胞比例）
# 作用：超过此比例标记为 BROKEN（大破损）
# 范围：0.05-0.3
bad_broken_area = 0.15

# 暗块角度覆盖阈值（度）
# 作用：紧凑度低的暗块，如果角度覆盖小于此值仍判为破损
#      暗环覆盖接近 360 度，月牙缺口通常只覆盖 60-150 度
# 范围：120-240
# 调整：月牙漏检 → 调高；暗环误判 → 调低
bad_blob_angle_span = 180


# ═══════════════════════════════════════
#  输出样式
# ═══════════════════════════════════════

# Top 细胞裁剪图范围倍数
# 作用：以细胞半径的此倍数为半径裁剪周边区域
# 范围：1.5-4.0
crop_pad = 2.0

# 描边线宽（像素）
# 范围：1-10
# 调整：大图上描边看不清 → 调高
contour_thickness = 2

# 标注文字大小
# 范围：0.3-2.0
font_scale = 0.7
